\chapter{Metodología}\label{chap:metodologia}

\begin{sloppypar}
\textbf{\textit{Fase 1:}} Revisión bibliográfica para la obtención del modelo no lineal del SLM.

\begin{enumerate}
  \item[a)] Análisis de las ecuaciones adaptándolas a la configuración del dispositivo.
  \item[b)] Simulación del sistema no lineal en los puntos de equilibrio.
\end{enumerate}

\textbf{\textit{Fase 2:}} Linealización alrededor de los puntos de equilibrio.

\begin{enumerate}
  \item[a)] Diseño y simulación del controlador lineal que estabilice, sin efecto atascamiento-deslizamiento, utilizando un controlador lineal con doble acción integral.
  \item[b)] Pruebas del controlador incluyendo el atascamiento-deslizamiento en el modelo lineal.
\end{enumerate}

\textbf{\textit{Fase 3:}} Aplicación del controlador al modelo no lineal.

\begin{enumerate}
  \item[a)] Pruebas del controlador en el modelo no lineal sin efecto atascamiento-deslizamiento.
  \item[b)] Pruebas del controlador en el modelo no lineal con efecto atascamiento-deslizamiento.
\end{enumerate}
\end{sloppypar}
